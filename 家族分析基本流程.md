
这个正则表达式 `'\.t1$'` 的含义如下：

- `\.` —— 匹配*字面意义的点号 `.`。
  在正则中 `.` 是特殊字符（表示"任意单个字符"），所以用反斜杠 `\` 转义后，它只匹配真正的 `.` 字符。
- `t1` —— 匹配字面字符序列 `t1`。
- `$` —— 行尾锚点，表示匹配必须位于字符串（或行）的末尾。

整体含义：匹配所有以 .t1 结尾的字符串。

原因：

- 避免 isoform 虚增 copy number
- 避免影响 CNV 分析

### 2.2 CYP基因分类

根据 Orthogroups.tsv的结果，将所有鉴定到的CYP基因分为core，softcore, dispensable几类，分类标准：

- core：所有基因组都有

- softcore：除core外，90%以上的基因组都有
- dispensable：除了12，大于1的基因组有
- private：只有一个基因组有

使用python脚本**ortho_pangenome_id_mapping.py**完成

```python
##ortho_pangenome_id_mapping.py
import pandas as pd
import argparse
import sys

def get_category(presence_count, n_species):
    """根据物种覆盖数判断分类"""
    if presence_count == n_species:
        return 'core'
    elif presence_count >= round(n_species * 0.9):
        return 'softcore'
    elif presence_count > 1:
        return 'dispensable'
    elif presence_count == 1:
        return 'private'
    else:
        return 'none'

def main():
    parser = argparse.ArgumentParser(description='Classify Orthofinder OGs and map gene IDs.')
    parser.add_argument('-c', '--count', required=True, help='Path to Orthogroups.GeneCount.tsv')
    parser.add_argument('-t', '--tsv', required=True, help='Path to Orthogroups.tsv')
    parser.add_argument('-o', '--output', required=True, help='Output filename')

    args = parser.parse_args()

    print("Loading files...")
    count_df = pd.read_csv(args.count, sep='\t')
    og_df = pd.read_csv(args.tsv, sep='\t')

    species_cols = [col for col in count_df.columns if col not in ['Orthogroup', 'Total']]
    n_species = len(species_cols)
    print(f"Detected {n_species} genomes.")


    presence_matrix = count_df[species_cols] > 0
    count_df['Category'] = presence_matrix.sum(axis=1).apply(lambda x: get_category(x, n_species))

    og_to_cat = dict(zip(count_df['Orthogroup'], count_df['Category']))

    print("Processing gene ID mapping...")
    results = []

    for genome in species_cols:
        cat_data = {
            'core': [],
            'softcore': [],
            'dispensable': [],
            'private': []
        }

        relevant_data = og_df[['Orthogroup', genome]].dropna() # 排除在该基因组中没有成员的 OG
        
        for _, row in relevant_data.iterrows():
            og_id = row['Orthogroup']
            genes = str(row[genome]).strip()
            
            if genes and genes != 'nan':
                category = og_to_cat.get(og_id, 'none')
                if category in cat_data:
                    gene_list = [g.strip() for g in genes.split(',')]
                    cat_data[category].extend(gene_list)


        for cat in ['core', 'softcore', 'dispensable', 'private']:
            all_genes = cat_data[cat]
            gene_str = ",".join(all_genes)
            gene_count = len(all_genes)
            
            results.append([genome, cat, gene_str, gene_count])

    final_df = pd.DataFrame(results, columns=['Genome', 'Category', 'GeneIDs', 'GeneCount'])
    final_df.to_csv(args.output, sep='\t', index=False)
    print(f"Analysis complete. Results saved to {args.output}")

if __name__ == "__main__":
    main()
```

## 三、系统发育和进化分析

整个泛家族中CYP基因数目过多，一起拿来建树是不合理的，这里的思路是提取各OG中最长序列作为代表，与拟南芥的CYP450基因构建进化树，获得每个OG的归类，同一OG内基因保守型高，因此，同一OG的基因应该是同一基因家族。

为了验证这一点，选择了三个OG内的所有序列分别与拟南芥的CYP450超家族基因建树，发现OG内序列在树上的分支与OG的分支一致，因此，根据OG建树的结果，将所有OG归入41个家族中，依据（Bak S, Beisson F, Bishop G, Hamberger B, Höfer R, Paquette S, Werck-Reichhart D. Cytochromes p450. Arabidopsis Book. 2011;9:e0144. doi: 10.1199/tab.0144. Epub 2011 Oct 6. PMID: 22303269; PMCID: PMC3268508.）将这41个家族进一步归入8个家族簇。

```python
#!/usr/bin/env python3
"""
从OrthoFinder的Orthogroup_Sequences目录中，
为每个OG提取代表序列（最长序列），以OG ID重命名，并合并为一个文件。
name：licongcong
用法：
    python extract_og_representatives.py \
        --og_dir /path/to/Orthogroup_Sequences \
        --output representative_cyps.faa \
        [--method longest]   # 可选: longest（默认）或 first
        [--min_len 50]       # 过滤掉长度不足的OG代表序列
        [--log extract.log]  # 可选：输出日志文件

输出：
    representative_cyps.faa   -- 所有OG代表序列，序列ID格式为 >OG0000001
    og_representative_map.tsv -- 对照表：OG_ID | 原始序列ID | 来源物种 | 序列长度
"""

import os
import argparse
import logging
from pathlib import Path


def parse_args():
    parser = argparse.ArgumentParser(
        description="从OrthoFinder Orthogroup_Sequences目录提取每个OG的代表序列"
    )
    parser.add_argument(
        "--og_dir", required=True,
        help="OrthoFinder输出的Orthogroup_Sequences目录路径"
    )
    parser.add_argument(
        "--output", default="representative_cyps.faa",
        help="输出的合并fasta文件路径（默认：representative_cyps.faa）"
    )
    parser.add_argument(
        "--method", choices=["longest", "first"], default="longest",
        help="代表序列选择策略：longest=最长序列（默认），first=第一条序列"
    )
    parser.add_argument(
        "--min_len", type=int, default=50,
        help="代表序列最短长度过滤阈值（默认：50 aa）"
    )
    parser.add_argument(
        "--log", default=None,
        help="日志输出文件路径（默认只输出到终端）"
    )
    return parser.parse_args()


def setup_logging(log_file=None):
    handlers = [logging.StreamHandler()]
    if log_file:
        handlers.append(logging.FileHandler(log_file, encoding="utf-8"))
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
        datefmt="%Y-%m-%d %H:%M:%S",
        handlers=handlers
    )


def parse_fasta(filepath):
    """
    简单的fasta解析器，不依赖BioPython。
    返回列表：[(header, sequence), ...]
    header是>之后的完整字符串，sequence是去除换行的序列字符串。
    """
    records = []
    header = None
    seq_parts = []

    with open(filepath, "r", encoding="utf-8") as f:
        for line in f:
            line = line.rstrip()
            if not line:
                continue
            if line.startswith(">"):
                if header is not None:
                    records.append((header, "".join(seq_parts)))
                header = line[1:]  # 去掉>
                seq_parts = []
            else:
                seq_parts.append(line)

    # 最后一条
    if header is not None:
        records.append((header, "".join(seq_parts)))

    return records


def get_species_from_header(header):
    """
    尝试从序列header中推断物种/hap来源。
    OrthoFinder的序列ID格式通常是：文件名前缀|原始ID 或 原始ID
    这里做简单提取，取第一个字段。
    """
    # 常见格式：SpeciesName_geneID 或 SpeciesName|geneID
    parts = header.split("|")
    if len(parts) > 1:
        return parts[0]
    parts = header.split("_")
    if len(parts) > 1:
        return parts[0]
    return "unknown"


def select_representative(records, method="longest"):
    """
    从一个OG的所有序列中选出代表序列。
    method: 'longest' 选最长序列；'first' 选第一条序列。
    返回 (header, sequence) 元组。
    """
    if not records:
        return None, None

    if method == "first":
        return records[0]

    # longest：选氨基酸序列最长的
    best_header, best_seq = records[0]
    for header, seq in records[1:]:
        if len(seq) > len(best_seq):
            best_header, best_seq = header, seq
    return best_header, best_seq


def main():
    args = parse_args()
    setup_logging(args.log)
    log = logging.getLogger()

    og_dir = Path(args.og_dir)
    if not og_dir.is_dir():
        log.error(f"找不到目录：{og_dir}")
        raise SystemExit(1)

    # 获取所有OG的fasta文件，按OG编号排序
    og_files = sorted(og_dir.glob("OG*.fa"))
    if not og_files:
        # 有些OrthoFinder版本输出.faa扩展名
        og_files = sorted(og_dir.glob("OG*.faa"))
    if not og_files:
        log.error(f"在 {og_dir} 中未找到任何 OG*.fa 或 OG*.faa 文件，请检查路径。")
        raise SystemExit(1)

    log.info(f"找到 {len(og_files)} 个OG文件，开始处理...")
    log.info(f"代表序列选择策略：{args.method}，最短长度过滤：{args.min_len} aa")

    # 对照表数据：(OG_ID, 原始序列ID, 物种, 序列长度)
    map_records = []

    # 统计信息
    total_og = 0
    skipped_empty = 0
    skipped_short = 0
    written = 0

    output_path = Path(args.output)
    map_path = output_path.parent / "og_representative_map.tsv"

    with open(output_path, "w", encoding="utf-8") as out_faa, \
         open(map_path, "w", encoding="utf-8") as out_map:

        # 写对照表表头
        out_map.write("OG_ID\t原始序列ID\t物种/来源\t序列长度(aa)\t成员总数\n")

        for og_file in og_files:
            og_id = og_file.stem  # 文件名去掉扩展名，如 OG0000001
            total_og += 1

            # 解析fasta
            records = parse_fasta(og_file)

            if not records:
                log.warning(f"{og_id}：文件为空，跳过。")
                skipped_empty += 1
                continue

            # 选代表序列
            rep_header, rep_seq = select_representative(records, method=args.method)

            # 长度过滤
            if len(rep_seq) < args.min_len:
                log.warning(
                    f"{og_id}：代表序列长度 {len(rep_seq)} aa < {args.min_len} aa，跳过。"
                    f"（原始ID：{rep_header[:60]}）"
                )
                skipped_short += 1
                continue

            # 以OG ID作为序列名写出
            out_faa.write(f">{og_id}\n")
            # 每行60个字符，标准fasta格式
            for i in range(0, len(rep_seq), 60):
                out_faa.write(rep_seq[i:i+60] + "\n")

            # 写对照表
            species = get_species_from_header(rep_header)
            out_map.write(
                f"{og_id}\t{rep_header}\t{species}\t{len(rep_seq)}\t{len(records)}\n"
            )

            written += 1

            # 每处理500个OG输出一次进度
            if written % 500 == 0:
                log.info(f"  已处理 {written} 个OG...")

    # 最终统计
    log.info("=" * 55)
    log.info(f"处理完成！")
    log.info(f"  OG文件总数     : {total_og}")
    log.info(f"  成功写出       : {written} 条代表序列")
    log.info(f"  跳过（空文件） : {skipped_empty} 个")
    log.info(f"  跳过（序列过短）: {skipped_short} 个")
    log.info(f"  输出fasta      : {output_path}")
    log.info(f"  对照表         : {map_path}")
    log.info("=" * 55)


if __name__ == "__main__":
    main()

```

```R
library(stringr)

# 读取数据（假设为制表符分隔）
table1 <- read.table("table1.txt", header=TRUE, sep="\t", stringsAsFactors=FALSE)
table2 <- read.table("table2.txt", header=TRUE, sep="\t", stringsAsFactors=FALSE)

# 步骤1：建立 OG → Family 映射
og_family_map <- table1 %>%
  select(OG_ID, Family) %>%
  distinct()

# 步骤2：合并数据，关联 Family
merged <- table2 %>%

merged <- merged %>%
  arrange(Genome, Family, OG)

# 步骤4：在每个 Genome × Family 内，按 OG 顺序给基因编号

merged <- merged %>%
  group_by(Genome, Family) %>%
  mutate(
    # 给每个OG在Family内分配一个顺序编号（不同OG得到不同编号块）
    OG_rank = dense_rank(OG),
    # 在整个 Genome × Family 内给每个基因一个全局序号
    Gene_index = row_number()
  ) %>%
```

```R
calc_density_from_gff <- function(gff_file, chr_len, window=WINDOW_SIZE) {

  message("  读取 GFF: ", basename(gff_file))

  # GFF 标准9列，跳过注释行
  gff <- read_tsv(
    gff_file,
    comment   = "#",
    col_names = c("seqname","source","feature",
                  "start","end","score","strand","frame","attr"),
    col_types = "ccciicccc",
    show_col_types = FALSE
  )

  # 只保留 gene 特征，且在目标染色体上
  target_chrs <- chr_len$Chr
  genes_all <- gff %>%
    filter(feature == "gene", seqname %in% target_chrs) %>%
    mutate(mid = as.integer((start + end) / 2))

  message("  GFF gene 数: ", nrow(genes_all))

  # 按染色体分窗口统计
  lapply(seq_len(nrow(chr_len)), function(i) {
    chr  <- chr_len$Chr[i]
    clen <- chr_len$End[i]
    brks <- c(seq(1, clen, by=window), clen + 1)
    cnt  <- hist(filter(genes_all, seqname==chr)$mid,
                 breaks=brks, plot=FALSE)$counts
    data.frame(
      Chr   = chr,
      Start = brks[-length(brks)],
      End   = pmin(brks[-1] - 1, clen),
      Value = cnt
    )
  }) %>% bind_rows()
}

          round(n_chr * w / 35.43307 * 0.85)

  ly <- if (!is.na(Ly_OVERRIDE)) Ly_OVERRIDE else 15
  list(width=w, Lx=lx, Ly=ly)
}
```

参考（Zhao W, Wang J, Yang H,etc Large-scale analysis of MYB genes in Cucurbitaceae identifies a novel gene regulating plant height. Hortic Res. 2025 Aug 15;12(11)）的方法，对orthofinder结果内每个OG中的每个hap内基因与其它hap中的所有基因配对。基因对的提取使用python脚本实现：

```python
#!/usr/bin/env python3
# generate_cross_hap_pairs.py

import pandas as pd
import itertools
from pathlib import Path

OG_TSV  = "/home/data/t110626/cyp/CYP_pep/OrthoFinder/Results_Feb27/Orthogroups/Orthogroups.tsv"
OUT_DIR = "/home/data/t110626/cyp/kaks/og_pairs_crosshap"

Path(OUT_DIR).mkdir(parents=True, exist_ok=True)

print("Loading Orthogroups.tsv...")
og_df = pd.read_csv(OG_TSV, sep='\t')
species_cols = [c for c in og_df.columns if c != 'Orthogroup']
hap_names    = [c.replace('.CYP.pep', '') for c in species_cols]

print(f"  {len(species_cols)} haplotypes, {len(og_df)} OGs")

total_pairs = 0

for _, row in og_df.iterrows():
    og = row['Orthogroup']

    # 每个hap的所有基因列表
    hap_genes = {}
    for sp, hap in zip(species_cols, hap_names):
        cell = str(row.get(sp, '')).strip()
        if cell and cell != 'nan':
            genes = [g.strip() for g in cell.split(',') if g.strip()]
            if genes:
                hap_genes[hap] = genes

    if len(hap_genes) < 2:
        continue

    # 跨hap两两配对：不同hap之间所有基因组合
    pairs = []
    hap_list = list(hap_genes.keys())
    for h1, h2 in itertools.combinations(hap_list, 2):
        for g1 in hap_genes[h1]:
            for g2 in hap_genes[h2]:
                pairs.append(f"{g1}\t{g2}")

    # 写入文件
    out_file = Path(OUT_DIR) / f"{og}.txt"
    with open(out_file, 'w') as f:
        f.write('\n'.join(pairs) + '\n')

    total_pairs += len(pairs)

print(f"Done! Total pairs: {total_pairs}")
print(f"OG pair files saved to: {OUT_DIR}")

# 看看最大最小的OG有多少对
sizes = [(f.stem, sum(1 for _ in open(f))) for f in Path(OUT_DIR).glob("*.txt")]
sizes.sort(key=lambda x: x[1], reverse=True)
print(f"\nTop 5 largest OGs:")
for og, n in sizes[:5]:
    print(f"  {og}: {n} pairs")
print(f"Smallest OG: {sizes[-1][0]}: {sizes[-1][1]} pairs")
```

```python
# Exon/intron count based on CDS segment count
import gzip, sys, glob

id_file = "../CYP_ids.oneIso.txt"
target_ids = set()
with open(id_file) as f:
    for line in f:
        tid = line.strip()
        if tid:
            target_ids.add(tid)
print(f"Loaded {len(target_ids)} target IDs", file=sys.stderr)

gff_files = sorted(
    glob.glob("*.gff.gz") + glob.glob("*.gff3.gz") +
    glob.glob("*.gff3")   + glob.glob("*.gff")
)
print(f"Found GFF files: {gff_files}", file=sys.stderr)

# mrna_info[tid] = {'cds': [(s,e),...], 'mrna_start': int, 'mrna_end': int}
mrna_info = {}

for gff_file in gff_files:
    opener = gzip.open if gff_file.endswith('.gz') else open
    print(f"Parsing {gff_file} ...", file=sys.stderr)
    with opener(gff_file, 'rt') as f:
        for line in f:
            if line.startswith('#'):
                continue
            parts = line.rstrip('\n').split('\t')
            if len(parts) < 9:
                continue
            ftype = parts[2]
            if ftype not in ('mRNA', 'transcript', 'CDS'):
                continue

            start = int(parts[3])
            end   = int(parts[4])
            attrs = parts[8]

            feat_id = parent = None
            for attr in attrs.split(';'):
                a = attr.strip()
                if a.startswith('ID='):
                    feat_id = a[3:].strip()
                elif a.startswith('Parent='):
                    parent = a[7:].strip()

            if ftype in ('mRNA', 'transcript'):
                if feat_id in target_ids:
                    if feat_id not in mrna_info:
                        mrna_info[feat_id] = {'cds': [], 'mrna_start': start, 'mrna_end': end}
                    else:
                        mrna_info[feat_id]['mrna_start'] = start
                        mrna_info[feat_id]['mrna_end']   = end

            elif ftype == 'CDS':
                if parent in target_ids:
                    if parent not in mrna_info:
                        mrna_info[parent] = {'cds': [], 'mrna_start': None, 'mrna_end': None}
                    mrna_info[parent]['cds'].append((start, end))

print(f"Matched {len(mrna_info)} mRNAs in GFF", file=sys.stderr)

with open("cyp.res.txt", 'w') as out:
    out.write("id\texon\tintron\tgene-len\tcds-len\tpep-len\n")
    missing = []
    for tid in sorted(target_ids):
        if tid not in mrna_info:
            missing.append(tid)
            continue
        info     = mrna_info[tid]
        cds_list = sorted(info['cds'])

        exon_count   = len(cds_list)           # exon数 = CDS段数
        intron_count = max(0, exon_count - 1)  # intron数 = CDS数 - 1

        if info['mrna_start'] is not None:
            gene_len = info['mrna_end'] - info['mrna_start'] + 1
        elif cds_list:
            gene_len = max(e[1] for e in cds_list) - min(e[0] for e in cds_list) + 1
        else:
            gene_len = 'NA'

        if cds_list:
            cds_len = sum(e[1] - e[0] + 1 for e in cds_list)
            pep_len = (cds_len - 3) // 3   # 减去终止密码子
        else:
            cds_len = pep_len = 'NA'

        out.write(f"{tid}\t{exon_count}\t{intron_count}\t{gene_len}\t{cds_len}\t{pep_len}\n")

    if missing:
        print(f"\nWarning: {len(missing)} IDs not found in any GFF:", file=sys.stderr)
        for m in missing:
            print(f"  {m}", file=sys.stderr)

print("Done → cyp.res.txt", file=sys.stderr)
```

```r
  zk_col <- paste0("zk_", t)

  zk_expr <- cyb_expr_mean[, zk_col]
  ac_expr <- cyb_expr_mean[, ac_col]
  
  # 避免log2(0)，加一个小值
  log2FC <- log2(zk_expr + 0.01) - log2(ac_expr + 0.01)

    GeneID  = rownames(cyb_expr_mean),
    log2FC  = log2FC,
    DAF     = paste0("DAF", t),

      log2FC >  log2FC_threshold ~ "Up in zk",
      log2FC < -log2FC_threshold ~ "Down in zk",

    )
  )
```
